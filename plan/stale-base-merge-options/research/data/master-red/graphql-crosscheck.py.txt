import json,subprocess
since="2026-11-06T22:05:00Z"
Q="""query($o:String!,$n:String!,$since:GitTimestamp!,$c:String){repository(owner:$o,name:$n){defaultBranchRef{target{... on Commit{history(since:$since,first:50,after:$c){pageInfo{hasNextPage endCursor} nodes{oid committedDate checkSuites(first:20){nodes{conclusion workflowRun{databaseId event workflow{name}} checkRuns(first:30,filterBy:{conclusions:[FAILURE,TIMED_OUT]}){nodes{name}}}}}}}}}}}"""
