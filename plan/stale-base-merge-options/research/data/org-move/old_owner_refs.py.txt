"""Count the old owner's references left in the 10 fleet repositories' committed master trees.

The org move's Definition of Done (note 005 §4) requires this count to be 0 after the move, so the
fleet never depends on GitHub's redirects. This is the ONE committed grep that assertion names.

Run:  python3 old_owner_refs.py.txt [--clones /data/projects] [--ref origin/master] [--no-fetch] [--json]

It reads the fleet list from livespec's `.livespec-fleet-manifest.jsonc` (its `fleet` entries only:
adopters stay out of the move) and runs `git grep` against each repository's `--ref` tree, so the
working tree, unpushed commits and untracked files never count. The DoD grades committed
CONFIGURATION, so these are excluded as history or narrative:
  - any `history/` directory (livespec spec revision snapshots);
  - any `archive/` directory (`plan/archive/`, livespec's root `archive/`);
  - live `plan/` research notes, which are narrative permalinks to PRs, commits and blobs. They move
    under `plan/archive/` when their plan closes, and the `prose` column counts them;
  - release-generated `CHANGELOG.md` files.

Graded columns (the DoD assertion):
  url   — `github.com/<old-owner>/<fleet-repo>` (https, ssh `github.com:` and API forms)
  ghcr  — `ghcr.io/<old-owner>`
Informational column (not graded by the draft DoD):
  slug  — a bare `<old-owner>/<fleet-repo>` NOT already inside a `url` hit: reusable-workflow
          `uses:` refs, marketplace `repo:` entries, `gh --repo` arguments. These also lean on the
          redirect after the move.
  prose — `url` + `ghcr` hits inside live `plan/` research notes (narrative links).
Exit status: 0 when every graded count is 0, 1 otherwise, 2 when a repository could not be read.
"""

from __future__ import annotations

import argparse
import json
import re
import subprocess
import sys
from pathlib import Path

HISTORY = (":(exclude,glob)**/history/**", ":(exclude,glob)**/archive/**", ":(exclude)archive",
           ":(exclude,glob)**/CHANGELOG.md")
EXCLUDES = (*HISTORY, ":(exclude)plan")
PROSE_ONLY = ("plan", *HISTORY)


def fleet(manifest: Path) -> tuple[str, list[str]]:
    text = re.sub(r"^\s*//.*$", "", manifest.read_text(), flags=re.M)
    data = json.loads(text)
    return data["owner"], [entry["repo"] for entry in data["fleet"]]


def grep(repo_dir: Path, ref: str, pattern: str, paths: tuple[str, ...] = (".", *EXCLUDES)) -> list[str]:
    proc = subprocess.run(
        ["git", "-C", str(repo_dir), "grep", "-I", "-o", "-P", pattern, ref, "--", *paths],
        capture_output=True,
        text=True,
        check=False,
    )
    if proc.returncode not in (0, 1):
        raise RuntimeError(proc.stderr.strip() or f"git grep exited {proc.returncode}")
    return [line for line in proc.stdout.splitlines() if line]


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--clones", default="/data/projects", type=Path)
    parser.add_argument("--ref", default="origin/master")
    parser.add_argument("--no-fetch", action="store_true")
    parser.add_argument("--json", action="store_true")
    args = parser.parse_args()

    owner, repos = fleet(args.clones / "livespec" / ".livespec-fleet-manifest.jsonc")
    names = "|".join(re.escape(r) for r in sorted(repos, key=len, reverse=True))
    end = r"(?![A-Za-z0-9_-])"
    url_re = rf"github\.com[:/](?:repos/)?{re.escape(owner)}/(?:{names}){end}"
    ghcr_re = rf"ghcr\.io/{re.escape(owner)}/"
    slug_re = rf"(?<![A-Za-z0-9_./:-]){re.escape(owner)}/(?:{names}){end}"

    rows, unreadable = [], []
    for repo in repos:
        repo_dir = args.clones / repo
        try:
            if not args.no_fetch:
                subprocess.run(["git", "-C", str(repo_dir), "fetch", "-q", "origin", "master"], check=True,
                               capture_output=True)
            head = subprocess.run(["git", "-C", str(repo_dir), "rev-parse", "--short", args.ref], check=True,
                                  capture_output=True, text=True).stdout.strip()
            url_hits = grep(repo_dir, args.ref, url_re)
            ghcr_hits = grep(repo_dir, args.ref, ghcr_re)
            slug_hits = grep(repo_dir, args.ref, slug_re)
            prose_hits = grep(repo_dir, args.ref, f"{url_re}|{ghcr_re}", PROSE_ONLY)
        except (subprocess.CalledProcessError, RuntimeError, FileNotFoundError) as err:
            unreadable.append({"repo": repo, "error": str(err)})
            continue
        files = {h.split(":", 2)[1] for h in url_hits + ghcr_hits}
        rows.append({"repo": repo, "ref": f"{args.ref}@{head}", "url": len(url_hits), "ghcr": len(ghcr_hits),
                     "slug": len(slug_hits), "prose": len(prose_hits), "graded_files": len(files)})

    graded = sum(r["url"] + r["ghcr"] for r in rows)
    if args.json:
        print(json.dumps({"owner": owner, "rows": rows, "unreadable": unreadable, "graded_total": graded}, indent=2))
    else:
        print(f"{'repository':36} {'ref':28} {'url':>5} {'ghcr':>5} {'files':>5} {'slug*':>6} {'prose*':>6}")
        for r in rows:
            print(f"{r['repo']:36} {r['ref']:28} {r['url']:>5} {r['ghcr']:>5} {r['graded_files']:>5} {r['slug']:>6} {r['prose']:>6}")
        for u in unreadable:
            print(f"{u['repo']:36} UNREADABLE: {u['error']}")
        print(f"graded total (url + ghcr): {graded}   * slug and prose are informational, not graded")
    if unreadable:
        return 2
    return 0 if graded == 0 else 1


if __name__ == "__main__":
    sys.exit(main())
