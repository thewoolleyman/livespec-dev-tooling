"""The org move's quiet check: is the fleet drained enough to start the cutover? (note 005 §2)

Run:  python3 quiet_check.py.txt [--owner thewoolleyman] [--halt-at <UTC ISO time>]
                                 [--org <org> --image <name>:<tag> ...] [--json]

Every condition is read LIVE, from GitHub through `gh` and from the gate cluster through
`with-gates-kubeconfig kubectl`. A condition this script cannot observe counts as NOT quiet: it fails
closed, so an outage can never read as a quiet fleet. The fleet is the `fleet` list in livespec's
`.livespec-fleet-manifest.jsonc`, read from `--clones`. The conditions, per repository:

  open-prs       every open PR carries `do-not-merge` and has auto-merge OFF
                 (`disarm-on-hold-label.yml` turns auto-merge off when the label is added)
  master-ci      the latest `ci.yml` run on master has completed, and its conclusion is success
  runs           no workflow run of any kind is `in_progress`, `queued`, `waiting` or `pending`,
                 except GHOSTS (see `is_ghost`). A ghost is a run whose status is `queued` (not `pending`
                 or `waiting`, which a concurrency group legitimately holds), that was never updated after it
                 was created (updated_at == created_at), that was created before `--halt-at` (the halt
                 announcement), and that EITHER has zero jobs OR is more than 24 h old at check time, with any
                 number of jobs. GitHub leaves such runs `queued` forever and refuses to cancel them. 31 were
                 measured on 2026-11-13: 30 with zero jobs, and livespec-runtime 32215471962, which has one job
                 and had not started in 86 days (comment on livespec-dev-tooling-xowhol, 00:47:15Z).
                 legacy-coordinator-agent ruled (b) for the zero-job case and (i) for the 24 h case. Ghosts
                 are reported as `ghost` with their ids and never deleted. Without `--halt-at`, nothing is
                 excluded, so the check still fails closed. `--self-test` checks the rule's boundary rows.
  release        no merged PR still carries `autorelease: pending` (Release Please has not tagged it)
  gate-jobs      no active gate Job for the repository in the cluster's `gates` namespace

and, once per invocation, for each `--image`:

  image          `ghcr.io/<org>/<name>` has a version tagged `<tag>` (note 005 §3 Phase 1 step 4)

Exit status: 0 when quiet, 1 when anything is not quiet or cannot be observed.
"""

from __future__ import annotations

import argparse
import datetime
import json
import re
import subprocess
import sys
from collections.abc import Callable
from pathlib import Path

ACTIVE_RUN_STATES = ("in_progress", "queued", "waiting", "pending", "requested")


class Unobservable(Exception):
    pass


def run_json(argv: list[str]) -> object:
    proc = subprocess.run(argv, capture_output=True, text=True, check=False)
    if proc.returncode != 0:
        raise Unobservable(f"{' '.join(argv[:4])}...: {(proc.stderr or proc.stdout).strip()[:200]}")
    try:
        return json.loads(proc.stdout or "null")
    except json.JSONDecodeError as err:
        raise Unobservable(f"{' '.join(argv[:4])}...: unparseable output ({err})") from err


def fleet(clones: Path) -> list[str]:
    text = re.sub(r"^\s*//.*$", "", (clones / "livespec" / ".livespec-fleet-manifest.jsonc").read_text(), flags=re.M)
    return [entry["repo"] for entry in json.loads(text)["fleet"]]


GHOST_MIN_AGE_SECONDS = 24 * 3600


def _utc(stamp: str) -> datetime.datetime:
    return datetime.datetime.fromisoformat(stamp.replace("Z", "+00:00"))


def is_ghost(*, status: str, created_at: str, updated_at: str, halt_at: str | None, now: datetime.datetime,
             jobs: Callable[[], int]) -> bool:
    """Ruling (b) plus ruling (i). `jobs` is called only when the 24 h rule alone does not decide."""
    if not halt_at or status != "queued" or updated_at != created_at or _utc(created_at) >= _utc(halt_at):
        return False
    if (now - _utc(created_at)).total_seconds() > GHOST_MIN_AGE_SECONDS:
        return True
    return jobs() == 0


SELF_TEST_ROWS = (
    # (description, status, age_seconds, updated_equals_created, created_before_halt, job_count, expected)
    ("24 h boundary, just over, with a job: ghost by (i)", "queued", 24 * 3600 + 1, True, True, 1, True),
    ("24 h boundary, exactly 24 h, with a job: active", "queued", 24 * 3600, True, True, 1, False),
    ("24 h boundary, just under, with a job: active", "queued", 24 * 3600 - 1, True, True, 1, False),
    ("young, zero jobs: ghost by (b)", "queued", 60, True, True, 0, True),
    ("old, but updated since created: active", "queued", 90 * 86400, False, True, 0, False),
    ("old, zero jobs, created after the halt: active", "queued", 90 * 86400, True, False, 0, False),
    ("pending, zero jobs (a concurrency group): active", "pending", 90 * 86400, True, True, 0, False),
)


def self_test() -> int:
    halt = datetime.datetime(2026, 11, 20, 10, 0, 0, tzinfo=datetime.timezone.utc)
    now = halt + datetime.timedelta(seconds=1)  # every row but the "after the halt" one is created before it
    failures = 0
    for desc, status, age, same, before, jobs, expected in SELF_TEST_ROWS:
        created = now - datetime.timedelta(seconds=age)
        if not before:
            created = halt + datetime.timedelta(seconds=1)
        stamp = created.strftime("%Y-%m-%dT%H:%M:%SZ")
        updated = stamp if same else (created + datetime.timedelta(seconds=5)).strftime("%Y-%m-%dT%H:%M:%SZ")
        got = is_ghost(status=status, created_at=stamp, updated_at=updated,
                       halt_at=halt.strftime("%Y-%m-%dT%H:%M:%SZ"), now=now, jobs=lambda j=jobs: j)
        ok = got == expected
        failures += not ok
        print(f"{'PASS' if ok else 'FAIL'}  {desc}: expected {expected}, got {got}")
    return 1 if failures else 0


def job_count(slug: str, run_id: int) -> int:
    data = run_json(["gh", "api", f"/repos/{slug}/actions/runs/{run_id}/jobs?per_page=1"])
    return int(data["total_count"])


def check_repo(slug: str, halt_at: str | None, now: datetime.datetime) -> dict[str, str]:
    out: dict[str, str] = {}

    def guard(name: str, fn) -> None:
        try:
            out[name] = fn()
        except Unobservable as err:
            out[name] = f"UNOBSERVABLE {err}"

    def open_prs() -> str:
        prs = run_json(["gh", "pr", "list", "-R", slug, "--state", "open", "--limit", "200",
                        "--json", "number,labels,autoMergeRequest"])
        bad = [f"#{p['number']}" + ("" if any(l["name"] == "do-not-merge" for l in p["labels"]) else "(unlabelled)")
               + ("(auto-merge on)" if p["autoMergeRequest"] else "")
               for p in prs
               if p["autoMergeRequest"] or not any(l["name"] == "do-not-merge" for l in p["labels"])]
        return "ok" if not bad else "NOT QUIET " + " ".join(bad)

    def master_ci() -> str:
        runs = run_json(["gh", "run", "list", "-R", slug, "--workflow", "ci.yml", "--branch", "master",
                         "--limit", "1", "--json", "databaseId,status,conclusion,headSha"])
        if not runs:
            return "UNOBSERVABLE no ci.yml run on master"
        r = runs[0]
        if r["status"] != "completed" or r["conclusion"] != "success":
            return f"NOT QUIET run {r['databaseId']} {r['status']}/{r['conclusion'] or '-'} @ {r['headSha'][:9]}"
        return f"ok {r['headSha'][:9]}"

    def runs() -> str:
        active, ghosts = [], []
        for state in ACTIVE_RUN_STATES:
            rows = run_json(["gh", "run", "list", "-R", slug, "--status", state, "--limit", "100",
                             "--json", "databaseId,workflowName,createdAt,updatedAt"])
            for r in rows:
                label = f"{r['databaseId']}:{r['workflowName']}:{state}"
                # ISO-8601 UTC strings in the same Z form compare correctly as text.
                if is_ghost(status=state, created_at=r["createdAt"], updated_at=r["updatedAt"], halt_at=halt_at,
                            now=now, jobs=lambda run_id=r["databaseId"]: job_count(slug, run_id)):
                    ghosts.append(label)
                else:
                    active.append(label)
        ghost_note = f"; ghost {' '.join(ghosts)}" if ghosts else ""
        return ("ok" if not active else "NOT QUIET " + " ".join(active)) + ghost_note

    def release() -> str:
        prs = run_json(["gh", "pr", "list", "-R", slug, "--state", "merged", "--label", "autorelease: pending",
                        "--limit", "20", "--json", "number"])
        return "ok" if not prs else "NOT QUIET release pending on " + " ".join(f"#{p['number']}" for p in prs)

    guard("open-prs", open_prs)
    guard("master-ci", master_ci)
    guard("runs", runs)
    guard("release", release)
    return out


def gate_jobs(repos: list[str]) -> dict[str, str]:
    try:
        data = run_json(["with-gates-kubeconfig", "kubectl", "get", "jobs", "-n", "gates", "-o", "json"])
    except Unobservable as err:
        return {repo: f"UNOBSERVABLE {err}" for repo in repos}
    active: dict[str, list[str]] = {repo: [] for repo in repos}
    for job in data["items"]:
        if (job.get("status", {}).get("active") or 0) <= 0:
            continue
        name = job["metadata"]["name"]
        # Job names are gate-<repo>-<suffix>; take the longest fleet name that matches.
        match = max((r for r in repos if name.startswith(f"gate-{r}-")), key=len, default=None)
        if match:
            active[match].append(name)
    return {repo: "ok" if not names else "NOT QUIET " + " ".join(names) for repo, names in active.items()}


def image_present(org: str, image: str) -> str:
    name, _, tag = image.partition(":")
    if not tag:
        return f"UNOBSERVABLE {image}: no :tag given"
    try:
        versions = run_json(["gh", "api", "--paginate", f"/orgs/{org}/packages/container/{name}/versions"])
    except Unobservable as err:
        return f"UNOBSERVABLE {err}"
    tags = {t for v in versions for t in v.get("metadata", {}).get("container", {}).get("tags", [])}
    return "ok" if tag in tags else f"NOT QUIET ghcr.io/{org}/{name}:{tag} absent"


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--owner", default="thewoolleyman")
    parser.add_argument("--clones", default="/data/projects", type=Path)
    parser.add_argument("--org")
    parser.add_argument("--image", action="append", default=[])
    parser.add_argument("--halt-at", help="halt announcement time, UTC, e.g. 2026-11-20T10:00:00Z")
    parser.add_argument("--json", action="store_true")
    parser.add_argument("--self-test", action="store_true", help="check the ghost rule's boundary rows and exit")
    args = parser.parse_args()
    if args.self_test:
        return self_test()
    if args.image and not args.org:
        parser.error("--image needs --org")
    if args.halt_at and not re.fullmatch(r"\d{4}-\d\d-\d\dT\d\d:\d\d:\d\dZ", args.halt_at):
        parser.error("--halt-at must be UTC in the form YYYY-MM-DDTHH:MM:SSZ")

    repos = fleet(args.clones)
    now = datetime.datetime.now(datetime.timezone.utc)
    results = {repo: check_repo(f"{args.owner}/{repo}", args.halt_at, now) for repo in repos}
    for repo, verdict in gate_jobs(repos).items():
        results[repo]["gate-jobs"] = verdict
    images = {image: image_present(args.org, image) for image in args.image}

    quiet = all(v.startswith("ok") for row in results.values() for v in row.values()) and all(
        v == "ok" for v in images.values())
    if args.json:
        print(json.dumps({"quiet": quiet, "repos": results, "images": images}, indent=2))
    else:
        for repo, row in results.items():
            for name, verdict in row.items():
                print(f"{repo:36} {name:10} {verdict}")
        for image, verdict in images.items():
            print(f"{'image':36} {image:10} {verdict}")
        print("QUIET" if quiet else "NOT QUIET")
    return 0 if quiet else 1


if __name__ == "__main__":
    sys.exit(main())
