"""Sample per-process CPU on this host and attribute it to test/gate load classes.

Every INTERVAL s, read utime+stime of every process, take the delta, and charge it to
a class derived from the process's own cwd/cmdline and its ancestors' cmdlines.
Writes one JSON line per interval: {"t":..., "total_ticks":..., "classes":{...}}.
"""
import json, os, sys, time

INTERVAL = float(sys.argv[1]); COUNT = int(sys.argv[2]); OUT = sys.argv[3]
HZ = os.sysconf("SC_CLK_TCK")


def read(pid):
    try:
        with open(f"/proc/{pid}/stat") as f:
            s = f.read()
        rest = s[s.rindex(")") + 2:].split()
        ppid = int(rest[1]); ticks = (int(rest[11]) + int(rest[12]), int(rest[13]) + int(rest[14]))
        with open(f"/proc/{pid}/cmdline", "rb") as f:
            cmd = f.read().replace(b"\0", b" ").decode(errors="replace")[:600]
        try:
            cwd = os.readlink(f"/proc/{pid}/cwd")
        except OSError:
            cwd = ""
        return ppid, ticks, cmd, cwd
    except (OSError, ValueError, IndexError):
        return None


def own_class(cmd, cwd):
    a = cmd.split()
    exe = os.path.basename(a[0]) if a else ""
    j = " ".join(a)
    if "/janitor" in cwd or "janitor-" in cwd or "janitor" in j[:300]:
        return "janitor-post-merge"
    if "covns" in j or "covns" in cwd:
        return "coverage-tree"
    if "red_green_replay" in j or "check-red-green-replay" in j:
        return "commit-red-green-replay"
    if exe == "lefthook" and "pre-push" in j:
        return "pre-push"
    if exe == "lefthook" and "pre-commit" in j:
        return "commit-pre-commit-aggregate"
    if exe == "git" and len(a) > 1 and a[1] in ("push",):
        return "pre-push"
    if exe == "git" and len(a) > 1 and a[1] in ("commit",):
        return "commit-pre-commit-aggregate"
    if "gate-run.sh __child" in j:
        return "adhoc-gate-start"
    return None


def classify(pid, procs):
    p = pid; first_test = None
    for _ in range(40):
        info = procs.get(p)
        if not info:
            break
        c = own_class(info[2], info[3])
        if c:
            return c
        j = info[2]
        if first_test is None and ("pytest" in j or "just check" in j or " just " in j or "just-check" in j or "coverage" in j):
            first_test = "adhoc-pytest-or-check"
        p = info[0]
        if p <= 1:
            break
    if first_test:
        return first_test
    info = procs.get(pid)
    j = info[2] if info else ""
    if any(x in j for x in ("k3s", "containerd", "kubelet", "dockerd")):
        return "k3s-docker"
    if "dolt" in j:
        return "dolt"
    if "herdr" in j:
        return "herdr"
    return "agents-and-other"


prev = {}
with open(OUT, "a") as out:
    for i in range(COUNT + 1):
        procs = {}
        for d in os.listdir("/proc"):
            if d.isdigit():
                r = read(int(d))
                if r:
                    procs[int(d)] = r
        if i:
            # ticks a vanished child already had charged while alive, by its parent pid
            vanished = {}
            for pid, (ppid, (own, cut), cmd) in prev.items():
                if pid not in procs or procs[pid][2][:80] != cmd:
                    vanished[ppid] = vanished.get(ppid, 0) + own + cut
            classes = {}; total = 0; top = {}
            for pid, (ppid, (own, cut), cmd, cwd) in procs.items():
                old = prev.get(pid)
                if not old or old[2] != cmd[:80]:
                    continue
                d = (own - old[1][0]) + max(0, (cut - old[1][1]) - vanished.get(pid, 0))
                if d <= 0:
                    continue
                c = classify(pid, procs)
                classes[c] = classes.get(c, 0) + d; total += d
                k = c + " :: " + cmd[:120]
                top[k] = top.get(k, 0) + d
            out.write(json.dumps({"t": time.time(), "interval": INTERVAL, "hz": HZ, "ncpu": os.cpu_count(),
                                  "total_ticks": total, "classes": classes,
                                  "top": dict(sorted(top.items(), key=lambda kv: -kv[1])[:25])}) + "\n")
            out.flush()
        prev = {pid: (v[0], v[1], v[2][:80]) for pid, v in procs.items()}
        if i < COUNT:
            time.sleep(INTERVAL)
